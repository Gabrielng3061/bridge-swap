pragma solidity ^0.8.20;

/**
 * @title BridgeSwap
 * @dev Contrato inteligente genÃ©rico e modular para gerir operaÃ§Ãµes de Bridge e Swap em redes EVM.
 */
interface IERC20 {
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
    function transfer(address recipient, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
}

contract BridgeSwap {
    address public owner;

    // Mapeamento para registar liquidez bloqueada por token
    mapping(address => uint256) public lockedLiquidity;

    event TokensBridged(
        address indexed sender,
        address indexed token,
        uint256 amount,
        uint256 targetChainId,
        uint256 timestamp
    );

    event TokensSwapped(
        address indexed user,
        address indexed tokenIn,
        address indexed tokenOut,
        uint256 amountIn,
        uint256 amountOut
    );

    modifier onlyOwner() {
        require(msg.sender == owner, "Apenas o proprietario pode executar esta acao");
        _;
    }

    constructor() {
        owner = msg.sender;
    }

    /**
     * @dev Bloqueia tokens na rede atual para posterior liberacao na rede de destino.
     */
    function bridgeTokens(address token, uint256 amount, uint256 targetChainId) external {
        require(amount > 0, "A quantidade deve ser maior que zero");

        IERC20 tokenContract = IERC20(token);

        bool success = tokenContract.transferFrom(msg.sender, address(this), amount);
        require(success, "Falha na transferencia dos tokens para a bridge");

        lockedLiquidity[token] += amount;

        emit TokensBridged(msg.sender, token, amount, targetChainId, block.timestamp);
    }

    /**
     * @dev Executa a troca de um token por outro disponivel no pool do contrato.
     */
    function swapTokens(address tokenIn, address tokenOut, uint256 amountIn, uint256 minAmountOut) external {
        require(amountIn > 0, "Quantidade de entrada invalida");

        IERC20(tokenIn).transferFrom(msg.sender, address(this), amountIn);

        uint256 amountOut = minAmountOut;

        require(IERC20(tokenOut).balanceOf(address(this)) >= amountOut, "Liquidez insuficiente para swap");
        IERC20(tokenOut).transfer(msg.sender, amountOut);

        emit TokensSwapped(msg.sender, tokenIn, tokenOut, amountIn, amountOut);
    }

    /**
     * @dev Permite ao owner retirar fundos de emergencia ou ajustar liquidez.
     */
    function emergencyWithdraw(address token, uint256 amount) external onlyOwner {
        IERC20(token).transfer(owner, amount);
    }
}

/**
 * @title TestToken
 * @dev Token ERC-20 completo para simulaÃ§Ã£o de liquidez, pontes e swaps.
 */
contract TestToken {
    string public name;
    string public symbol;
    uint8 public decimals = 18;
    uint256 public totalSupply;

    address public owner;

    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;

    event Transfer(address indexed from, address indexed to, uint256 value);
    event Approval(address indexed owner, address indexed spender, uint256 value);

    modifier onlyOwner() {
        require(msg.sender == owner, "Apenas o proprietario pode cunhar tokens");
        _;
    }

    constructor(string memory _name, string memory _symbol, uint256 initialSupply) {
        name = _name;
        symbol = _symbol;
        owner = msg.sender;

        // Cunhar suprimento inicial para o criador
        mint(msg.sender, initialSupply * 10**uint256(decimals));
    }

    function transfer(address to, uint256 amount) public returns (bool) {
        require(balanceOf[msg.sender] >= amount, "Saldo insuficiente");
        balanceOf[msg.sender] -= amount;
        balanceOf[to] += amount;
        emit Transfer(msg.sender, to, amount);
        return true;
    }

    function approve(address spender, uint256 amount) public returns (bool) {
        allowance[msg.sender][spender] = amount;
        emit Approval(msg.sender, spender, amount);
        return true;
    }

    function transferFrom(address from, address to, uint256 amount) public returns (bool) {
        require(balanceOf[from] >= amount, "Saldo insuficiente");
        require(allowance[from][msg.sender] >= amount, "Allowance/Aprovacao insuficiente");

        balanceOf[from] -= amount;
        allowance[from][msg.sender] -= amount;
        balanceOf[to] += amount;

        emit Transfer(from, to, amount);
        return true;
    }

    /**
     * @dev FunÃ§Ã£o para criar novos tokens de teste sob demanda.
     */
    function mint(address to, uint256 amount) public onlyOwner {
        totalSupply += amount;
        balanceOf[to] += amount;
        emit Transfer(address(0), to, amount);
    }
}

import logging
from web3 import Web3
from utils.config import NETWORKS, PRIVATE_KEY

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(name)s: %(message)s")
logger = logging.getLogger("BlockchainConnection")


class BlockchainConnection:
    """Gerencia a conexÃ£o com qualquer rede EVM."""

    def __init__(self, network_name: str) -> None:
        if network_name not in NETWORKS:
            raise ValueError(f"Rede {network_name} nÃ£o configurada no config.py")

        self.rpc_url = NETWORKS[network_name]["rpc"]
        self.private_key = PRIVATE_KEY
        self.w3 = self._initialize_connection(network_name)

    def _initialize_connection(self, network_name: str) -> Web3:
        logger.info(f"A conectar Ã  rede {network_name}...")
        w3 = Web3(Web3.HTTPProvider(self.rpc_url))

        if not w3.is_connected():
            logger.error(f"Falha ao conectar Ã  rede {network_name}.")
            raise ConnectionError(f"Sem ligaÃ§Ã£o a {network_name}.")

        logger.info(f"ConexÃ£o a {network_name} estabelecida!")
        return w3

    def get_web3(self) -> Web3:
        return self.w3

install_solc("0.8.20")


def compile_contract(file_path: str, contract_name: str):
    logger.info(f"A compilar o contrato {contract_name}...")
    with open(file_path, "r", encoding="utf-8") as f:
        source_code = f.read()

    compiled_sol = compile_standard(
        {
            "language": "Solidity",
            "sources": {file_path: {"content": source_code}},
            "settings": {
                "outputSelection": {
                    "*": {"*": ["abi", "evm.bytecode"]}
                }
            },
        },
        solc_version="0.8.20",
    )

    bytecode = compiled_sol["contracts"][file_path][contract_name]["evm"]["bytecode"]["object"]
    abi = compiled_sol["contracts"][file_path][contract_name]["abi"]
    return abi, bytecode


def deploy_contract(network_name: str, abi, bytecode, *constructor_args):
    conn = BlockchainConnection(network_name)
    w3 = conn.get_web3()
    account = w3.eth.account.from_key(conn.private_key)

    contract = w3.eth.contract(abi=abi, bytecode=bytecode)

    logger.info(f"A enviar transaÃ§Ã£o de deploy para a rede: {network_name}...")
    tx = contract.constructor(*constructor_args).build_transaction({
        "from": account.address,
        "nonce": w3.eth.get_transaction_count(account.address),
        "gasPrice": w3.eth.gas_price,
    })

    signed_tx = w3.eth.account.sign_transaction(tx, conn.private_key)
    tx_hash = w3.eth.send_raw_transaction(signed_tx.raw_transaction)
    logger.info(f"TransaÃ§Ã£o enviada! Hash: {tx_hash.hex()}")

    tx_receipt = w3.eth.wait_for_transaction_receipt(tx_hash)
    logger.info(f"[âœ“] Contrato implantado com sucesso em: {tx_receipt.contractAddress}")
    return tx_receipt.contractAddress


def run_full_deploy(target_network: str):
    logger.info(f"=== INICIANDO DEPLOY COMPLETO NA REDE {target_network} ===")

    # 1. Deploy do Token ERC-20 de Teste

                amount = event["args"]["amount"]

            time.sleep(3)
        except Exception as e:
            logger.error(f"Erro na escuta de blocos: {e}")
            time.sleep(5)

def simulate_bridge_and_swap(network_name: str, token_addr: str, bridge_addr: str, amount_tokens: float):
    conn = BlockchainConnection(network_name)
    w3 = conn.get_web3()
    account = w3.eth.account.from_key(conn.private_key)

    token_contract = w3.eth.contract(address=w3.to_checksum_address(token_addr), abi=TOKEN_ABI)
    bridge_contract = w3.eth.contract(address=w3.to_checksum_address(bridge_addr), abi=BRIDGE_ABI)

    amount_wei = w3.to_wei(amount_tokens, "ether")

    # 1. Passo: AprovaÃ§Ã£o ERC-20 (Approve)
    logger.info(f"1. A aprovar {amount_tokens} pTEST para o contrato de Bridge...")
    approve_tx = token_contract.functions.approve(bridge_addr, amount_wei).build_transaction({
        "from": account.address,
        "nonce": w3.eth.get_transaction_count(account.address),
        "gasPrice": w3.eth.gas_price,
    })
    signed_approve = w3.eth.account.sign_transaction(approve_tx, conn.private_key)
